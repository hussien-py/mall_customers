X = df[
    [

        'Spending Score (1-100)'
    ]
]

FEATURE SCALING¶

In [261]:
# Standardize the selected features so each has mean 0 and standard deviation 1.
# Without this, the feature with the larger numeric range can dominate distance-based clustering.
scaler = StandardScaler()  # Creates the scaler.

# fit_transform learns each column's mean and standard deviation from X,
# then returns the scaled values as a NumPy array (column names are dropped).
X_scaled = scaler.fit_transform(X)

FIND OPTIMAL K¶

In [262]:
# Lists to store the inertia and silhouette score for each value of K
inertia_values = []
silhouette_values = []

# Test K from 2 to 10 (silhouette needs at least 2 clusters)
k_values = range(2, 11)

for k in k_values:

    # Fit K-Means with k clusters
    # random_state makes results reproducible, n_init=10 runs 10 random starts and keeps the best
    model = KMeans(
        n_clusters=k,
        random_state=42,
        n_init=10
    )

    labels = model.fit_predict(X_scaled)

    # Inertia is the within-cluster sum of squared distances (lower is tighter)
    inertia_values.append(model.inertia_)

    # Silhouette score measures separation between clusters (closer to 1 is better)
    silhouette_values.append(
        silhouette_score(X_scaled, labels)
    )


# Elbow Method
# Look for the bend in the curve where adding clusters stops helping much
plt.figure(figsize=(8, 5))

plt.plot(
    list(k_values),
    inertia_values,
    marker='o'
)

plt.xlabel("Number of Clusters (K)")
plt.ylabel("Inertia")
plt.title("Elbow Method")

plt.grid()
plt.show()


# Silhouette Score
# Look for the K with the highest score
plt.figure(figsize=(8, 5))

plt.plot(
    list(k_values),
    silhouette_values,
    marker='o'
)

plt.xlabel("Number of Clusters (K)")

columns = [
    'Age',
    'Annual Income (k$)',
    'Spending Score (1-100)'
]

for column in columns:

    Q1 = df[column].quantile(0.25)

    Q3 = df[column].quantile(0.75)

    IQR = Q3 - Q1

    lower = Q1 - 1.5 * IQR

    upper = Q3 + 1.5 * IQR

    outliers = df[
        (df[column] < lower) |
        (df[column] > upper)
    ]

    print(f"\n{column} Outliers:")

    print("Number of Outliers:", len(outliers))

    print("Lower Bound:", lower)

    print("Upper Bound:", upper)
